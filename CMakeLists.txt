cmake_minimum_required(VERSION 3.16)
project(ChatConnect LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PostgreSQL REQUIRED)
add_executable(ChatConnectWeb server.cpp)
target_link_libraries(ChatConnectWeb PRIVATE Threads::Threads PostgreSQL::PostgreSQL OpenSSL::Crypto)
add_executable(ChatConnectTests tests/auth_tests.cpp)
target_link_libraries(ChatConnectTests PRIVATE Threads::Threads PostgreSQL::PostgreSQL OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(ChatConnectWeb PRIVATE ws2_32)
  target_link_libraries(ChatConnectTests PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME ChatConnectAuthBasics COMMAND ChatConnectTests)
